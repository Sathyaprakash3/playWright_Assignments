/**
 * 🚫 DO NOT DO
 */

export const doNotDo = {
  // ❌ WRONG: Using @playwright/test
  'do NOT use': `import { test } from '@playwright/test'`,
  'do INSTEAD': `import { test } from '../../fixtures/customTest'`,
  
  // ❌ WRONG: test.only blocks other tests
  'do NOT use': `test.only('some test', ...)`,
  'do INSTEAD': `test('some test', ...)`,
  
  // ❌ WRONG: Hardcoding URLs
  'do NOT use': `await page.goto('https://www.leafground.com/')`,
  'do INSTEAD': `await page.goto(testData.baseUrl)`,
  
  // ❌ WRONG: Shared state between tests
  'do NOT use': `let sharedVar = 0; test('t1', () => { sharedVar++ })`,
  'do INSTEAD': `test('t1', () => { const localVar = 1 })`,
  
  // ❌ WRONG: Test depends on another test
  'do NOT use': `test('test 2', async () => { expect(resultFromTest1).toBe(...) })`,
  'do INSTEAD': `test('test 2', async () => { setup own data; test independently })`,
  
  // ❌ WRONG: Using test.beforeAll in test file
  'do NOT use': `test.beforeAll(() => { globalSetup() })`,
  'do INSTEAD': `// Use fixtures or global beforeEach in testIsolationHooks.ts`,
  
  // ❌ WRONG: Hardcoding test data
  'do NOT use': `expect(element).toHaveText('INSTOCK')`,
  'do INSTEAD': `expect(element).toHaveText(testData.table.expectedStatus)`,
  
  // ❌ WRONG: No wait for page load
  'do NOT use': `await page.goto(url); await action()`,
  'do INSTEAD': `await page.goto(url); await page.waitForLoadState('networkidle'); await action()`,
};
